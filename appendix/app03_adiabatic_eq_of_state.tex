\section{Derivation and motivation of adiabatic equation of state}
\label{sec: App. Adiabatic Equation of State}
 

We show that for an ideal gas, conserving entropy along fluid trajectories is
equivalent to the relation $\dfrac{d}{dt}(P \varrho^{-\gamma})=0$.
 
\subsection*{First Law of Thermodynamics}
 
Per unit mass, the first law reads
\begin{equation}
    de = T\,ds - P\,d\!\left(\frac{1}{ \varrho}\right),
    \label{eq:firstlaw}
\end{equation}
where $e$ is specific internal energy (per unit mass), $s$ is specific entropy (per unit mass), $1/ \varrho = 1/ n m$
is specific volume (per unit mass), $n$ is number density and $m$ is mass per particle.
 
\subsection*{Ideal Gas Relations}
 
For an ideal gas two relations hold:
\begin{align}
    e &= c_v T, \label{eq:ig1}\\
    P &= \frac{k_B}{m}\, \varrho T, \label{eq:ig2}
\end{align}
where $c_v$ is the specific heat at constant volume, $k_B$ is Boltzmann's
constant, and $m$ is the particle mass.
 
\subsection*{Differential of Entropy}
 
Substituting \eqref{eq:ig1} and \eqref{eq:ig2} into \eqref{eq:firstlaw} gives
\begin{equation}
    ds = c_v\,\frac{dT}{T} - \frac{k_B}{m}\,\frac{d \varrho}{ \varrho}.
    \label{eq:ds}
\end{equation}
Integrating:
\begin{equation}
    s = c_v \ln T - \frac{k_B}{m}\ln \varrho + \text{const}.
    \label{eq:s_T}
\end{equation}
 
\subsection*{Eliminating Temperature}
 
From \eqref{eq:ig2}, $T = Pm/( \varrho k_B)$, so $\ln T = \ln P - \ln \varrho +
\text{const}$. Substituting into \eqref{eq:s_T}:
\begin{equation*}
    s = c_v\ln\!\left(\frac{P}{ \varrho}\right) - \frac{k_B}{m}\ln \varrho + \text{const}.
\end{equation*}
Expanding and collecting the $\ln \varrho$ terms:
\begin{equation*}
    s = c_v\ln P - \left(c_v + \frac{k_B}{m}\right)\ln \varrho + \text{const}.
\end{equation*}
Using the ideal-gas identity $c_p - c_v = k_B/m$ and the definition
$\gamma \equiv c_p/c_v$, we have $c_v + k_B/m = c_p = \gamma c_v$, so
\begin{equation}
    s = c_v\!\left(\ln P - \gamma\ln \varrho\right) + \text{const}
      = c_v\ln\!\left(P \varrho^{-\gamma}\right) + \text{const}.
    \label{eq:s_final}
\end{equation}
 
\subsection*{Adiabatic Condition}
 
Since $c_v > 0$, equation \eqref{eq:s_final} shows that
\begin{equation*}
    \frac{ds}{dt} = 0
    \quad\iff\quad
    \frac{d}{dt}\!\left(P \varrho^{-\gamma}\right) = 0,
\end{equation*}
where $d/dt = \partial_t + \mathbf{v}\cdot\nabla$ is the material derivative
following a fluid element. This is the \emph{adiabatic equation of state}, it simply states 
 that entropy is conserved along fluid trajectories.
 

\subsection*{Note: Plasma is not an ideal gas}
The derivation relies on the ideal gas law \eqref{eq:ig2}. In a plasma context this assumption should
be revisited, as inter-particle interactions and anisotropic pressure can modify the equation of state.
