\chapter{Static Equilibrium Model in Cylindrical Coordinates under Axisymmetric Assumptions}
\label{chap: Static equilibrium model in cylindrical coordinates under axisymmetric assumptions}

\section{Axisymmetric and Poloidal Formulation}
\label{sec: equlibrium Axisymmetric and poloidal formulation}


In the following we will assume the magnetic configuration to be axisymmetric, i.e. independent of the toroidal angle $\varphi$.

The magnetic field thus simplifies to
\begin{equation}
    \vec{B} = B_r (r,z) \hat{e}_r + B_z(r,z) \hat{e}_z + B_\varphi(r,z) \hat{e}_\varphi \, .
    \label{eq: axisymm. magn. field}
\end{equation}
.

In cylindrical coordinates $(r, \varphi, z)$, the gradient, divergence and curl are:

\begin{align*}
    \nabla  f &= \frac{\partial f}{\partial r} \hat{e}_r + \frac{1}{r}\frac{\partial f}{\partial \varphi} \hat{e}_\varphi + \frac{\partial f}{\partial z} \hat{e}_z  \, ,\\
    \nabla \cdot \vec{F} &= \frac{1}{r}\frac{\partial (r F_r)}{\partial r} + \frac{1}{r}\frac{\partial F_\varphi}{\partial \varphi} + \frac{\partial F_z}{\partial z}  \, ,\\
    \nabla \times \vec{F} &= \left(\frac{1}{r}\frac{\partial F_z}{\partial \varphi} - \frac{\partial F_\varphi}{\partial z}\right)\hat{e}_r
    + \left(\frac{\partial F_r}{\partial z} - \frac{\partial F_z}{\partial r}\right)\hat{e}_\varphi
    + \frac{1}{r}\left(\frac{\partial (r F_\varphi)}{\partial r} - \frac{\partial F_r}{\partial \varphi}\right)\hat{e}_z \, ,
\end{align*}

so
\begin{align}
    \nabla \cdot \vec{B}  &= \frac{1}{r}\frac{\partial (r B_r)}{\partial r} + \frac{\partial B_z}{\partial z} \, , \label{eq: axisymm. cylindrical div}\\
    \nabla \times \vec{B} &= 
\left(\frac{\partial B_r}{\partial z} - \frac{\partial B_z}{\partial r}\right)\hat{e}_\varphi 
- \frac{\partial B_\varphi}{\partial z}\,\hat{e}_r 
+ \frac{1}{r}\frac{\partial(r B_\varphi)}{\partial r}\,\hat{e}_z  \, .
    \label{eq: axisymm. cylindrical curl}
\end{align}

In our axisymmetric approximation we disregard any impact the tunnels have. Since the polomac does not include any toroidal field coils,
the only possible source of a nonzero $B_\varphi(r,z) \hat{e}_\varphi$ are these tunnels, which we are disregarding (so zeroth order in our poloidal expansion).

We thus set $B_\varphi(r,z) \hat{e}_\varphi \equiv 0$, which further simplifies \eqref{eq: axisymm. magn. field} to
\begin{equation}
    \vec{B} = B_r (r,z) \hat{e}_r + B_z(r,z) \hat{e}_z  \, ,
    \label{eq: axisymm. pol. magn. field}
\end{equation}

and \eqref{eq: axisymm. cylindrical curl} to 
\begin{align}
    \nabla \times \vec{B} &= \left(\frac{\partial B_r}{\partial z} - \frac{\partial B_z}{\partial r}\right)\hat{e}_\varphi   \, .
    \label{eq: poloidal cylindrical curl}
\end{align}



\section{Static Equilibrium}

For the following derivation we closely work with Chapter 6 of \cite{Freidberg2014} which derives the Grad-Shafranov equation.
Due to our special geometry, we can make a couple of simplifications.

In case of a static equilibrium ($\vec{V}=0$, $\frac{\partial}{\partial t} \cdot =0$), 
of the ideal MHD equations only the following \emph{static equilibrium equations} remain.

\begin{itemize}
    \item the static momentum equation:
            \begin{equation}
                 \vec{J} \times \vec{B} = \nabla P
                 \label{eq: static momentum eq.}
            \end{equation}

    \item static Maxwell equations:
        \begin{itemize}
            \item 
                \begin{equation}
                    \nabla\times\vec{B} =  \mu_0 \vec{J}
                \end{equation}
            \item
                \begin{equation}
                    \nabla\cdot\vec{B} = 0
                \end{equation}
        \end{itemize}
\end{itemize}



Using \eqref{eq: axisymm. cylindrical div}, \eqref{eq: axisymm. pol. magn. field} and \eqref{eq: poloidal cylindrical curl} 
we can rewrite these equations and only the following components remain:

\begin{itemize}
    \item $\nabla \cdot \vec{B} = 0$ becomes:
        \begin{equation}
            \frac{1}{r}\frac{\partial (r B_r)}{\partial r} + \frac{\partial B_z}{\partial z} = 0
            \label{eq: divB}
        \end{equation}

    \item Ampère's law $\nabla \times \vec{B} = \mu_0 \vec{J}$ gives a purely toroidal current density:
        \begin{equation}
            \vec{J} = J_\varphi(r,z)\,\hat{e}_\varphi, \qquad 
            J_\varphi = \frac{1}{\mu_0}\left(\frac{\partial B_r}{\partial z} - \frac{\partial B_z}{\partial r}\right)
            \label{eq: J_phi}
        \end{equation}

    \item The momentum equation $\vec{J} \times \vec{B} = \nabla P$ yields
        \begin{align}
            J_\varphi\,\hat{e}_\varphi \times \left(B_r\,\hat{e}_r + B_z\,\hat{e}_z\right) 
            &= \frac{\partial P}{\partial r}\hat{e}_r + \frac{\partial P}{\partial z}\hat{e}_z  \, .
            \label{eq: mom eq}
        \end{align}
        Using $\hat{e}_\varphi \times \hat{e}_r = -\hat{e}_z$ and $\hat{e}_\varphi \times \hat{e}_z = \hat{e}_r$, 
        this gives two scalar equations:
        \begin{align}
            J_\varphi B_z &= \frac{\partial P}{\partial r} \label{eq: momentum_r}\\
            -J_\varphi B_r &= \frac{\partial P}{\partial z} \label{eq: momentum_z}
        \end{align}
\end{itemize}

Notice that \eqref{eq: divB} implies that $\vec{B}$ can be written identically
by introducing the poloidal flux function $\psi(r,z)$ such that
\begin{equation}
    \vec{B} = \frac{1}{r} \nabla\psi \times \hat{e}_\varphi,
    \qquad
    B_r = -\frac{1}{r}\frac{\partial \psi}{\partial z}, \quad
    B_z = \frac{1}{r}\frac{\partial \psi}{\partial r}.
    \label{eq: B as fct of psi}
\end{equation}


By construction, $\vec{B} \cdot \nabla\psi = 0$, so $\psi$ is constant
along field lines.

Dotting $\vec{B}$ into the momentum equation \eqref{eq: static momentum eq.}
gives $\vec{B} \cdot \nabla P = 0$, so $P$ is constant along field lines as well.

We assume that in the plasma domain the field lines form closed, nested loops
around the torus of central coils, consistent with a dipole-like topology, but not
derived from ideal MHD. Under this assumption each field line loop
has a unique value of $\psi$, making $\psi$ a valid flux surface label
and $P(\psi)$ a well-defined single-valued function.

The function $P(\psi)$ cannot be determined from ideal MHD alone. As
\cite{Freidberg2014} notes, transport theory, experimental data, or physical
intuition are needed to specify it. To obtain a fully determined system,
resistive MHD would have to be used.


\paragraph{Remark} If we did not assume $B_\varphi \equiv 0$, we would have restricted 
the above discussion to $B_\text{pol}$ and added a similar function for $B_\varphi$ 
(see \cite{Freidberg2014}). The argument for nested flux surfaces would get more involved.



\subsection{Reduced Grad-Shafranov Equation}

We can rewrite \eqref{eq: J_phi} as
\begin{equation}
    \mu_0 J_\varphi = - \frac{1}{r} \Delta^* \psi  \, ,
    \label{eq: J_phi as delta*}
\end{equation}
where, using cylindrical divergence again, 
\begin{equation*}
    \Delta^* \psi \coloneq r^2 \nabla \cdot \left(\frac{\nabla \psi}{r^2}\right) = r \frac{\partial}{\partial r}\left(\frac{1}{r}\frac{\partial \psi}{\partial r}\right) + \frac{\partial^2 \psi}{\partial z^2}  \, .
\end{equation*}

If we furthermore use the chain rule ($\nabla P = \frac{dP}{d\psi} \nabla \psi$) and \eqref{eq: J_phi as delta*}, 
we see that for nontrivial $\psi(r,z)$ both components of the momentum equation (\eqref{eq: momentum_r}, \eqref{eq: momentum_z})
are satisfied simultaneously if and only if 
\begin{equation}
    \label{eq: reduced grad-shafranov eq}
    \Delta^*\psi = - \mu_0 r^2 \frac{dP}{d\psi}.
\end{equation}
We call this the \emph{reduced Grad-Shafranov equation}. 

\paragraph{Remark} Instead of specifying an experimentally obtained pressure profile
$P(\psi)$, we could also specify an experimentally obtained current profile $J_\varphi(r,z)$, insert it in 
\eqref{eq: J_phi as delta*} and recover the pressure via \eqref{eq: B as fct of psi}
and the static momentum equation \eqref{eq: static momentum eq.}. 
This does not change the fact, that ideal MHD is underdetermined.

\subsection{Boundary Conditions}
\label{subsec: equilibrium BC}

\subsubsection{Geometric Axis ($r = 0$)}
For $\vec{B}$ and $p$ to remain finite on the symmetry axis, we require
\begin{equation*}
    \nabla\psi \xrightarrow{r \to 0} 0.
\end{equation*}
This is a regularity condition, not a physical wall.

\subsubsection{Conducting Walls (Inner Coil Surface and Outer Vacuum Vessel)}
Assuming the wall materials are perfect conductors and initially there is no magnetic field inside the wall, 
the integral version of Gauss' law for magnetism \eqref{eq:gauss_b} gives 

\begin{equation}
    \vec{B} \cdot \hat{n} = 0 \quad \text{on all walls}
    \label{eq: BC wall}
\end{equation}
In the static equilibrium ($\partial_t B = 0$) this holds for all times.

In our axisymmetric setup, the wall normal $\hat{n} = n_r\hat{e}_r + n_z\hat{e}_z$ has no
$\varphi$-component. We can recognize the wall tangent in the poloidal plane $\hat{t} = \hat{e}_\varphi \times \hat{n}$
when we insert \eqref{eq: B as fct of psi} into \eqref{eq: BC wall}
and use the triple product
\begin{equation*}
    0 = \left(\nabla\psi \times \hat{e}_\varphi\right)\cdot\hat{n}
    = \hat{t}\cdot \nabla\psi  \, ,
\end{equation*}
and thus $\psi$ is constant along the walls in the poloidal plane.
Because $\psi$ is only a function of $r,z$, it is thus constant on the entire inner and outer wall, respectively.
